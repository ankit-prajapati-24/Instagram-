# Voice: local hi-IN neural TTS, deliberately not routed through OmniRoute.
# Piper is the default engine -- offline, MIT-licensed, chosen by ear over
# twelve edge-tts voices. It downloads its ~61 MB voice model on first use.
piper-tts

# edge-tts is the fallback engine, used when RAHASYA_VOICE_ENGINE=edge or
# when Piper cannot synthesise.
edge-tts

# Caption word alignment, OFF by default (RAHASYA_ALIGN=1 turns it on).
# Used as a forced aligner, not a transcriber: it measures where each word
# lands in the audio we just synthesised, instead of interpolating caption
# words across the beat by character count. CPU-only -- ctranslate2 ships a
# cp314 wheel, so no torch is pulled in. The `base` model downloads ~40MB
# from Hugging Face on FIRST USE, which is why the switch defaults to off.
faster-whisper
